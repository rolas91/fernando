function numericHours(value: unknown) {
  const parsed = typeof value === 'number' ? value : Number(value);
  return Number.isFinite(parsed) ? parsed : 0;
}

function clockMinutes(value?: string) {
  const match =
    value?.trim().match(/^(\d{1,2}):(\d{2})\s*([AP])\.?M\.?$/i) ??
    value?.trim().match(/^(\d{1,2}):(\d{2})$/);
  if (!match) return null;
  let hours = Number(match[1]);
  if (match[3]?.toUpperCase() === 'A' && hours === 12) hours = 0;
  if (match[3]?.toUpperCase() === 'P' && hours !== 12) hours += 12;
  return hours * 60 + Number(match[2]);
}

function defaultDateRange() {
  const end = new Date();
  const start = new Date(end);
  start.setDate(end.getDate() - 29);
  return {
    start: dateInputValue(start),
    end: dateInputValue(end),
  };
}

function dateInputValue(date: Date) {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const day = String(date.getDate()).padStart(2, '0');
  return `${year}-${month}-${day}`;
}

function startOfWeek(date: Date) {
  const start = new Date(date);
  start.setDate(date.getDate() - date.getDay());
  return start;
}

function startOfMonth(date: Date) {
  return new Date(date.getFullYear(), date.getMonth(), 1);
}

function dateKey(value?: string) {
  if (!value) return '';
  if (/^\d{4}-\d{2}-\d{2}/.test(value)) return value.slice(0, 10);
  const parsed = new Date(value);
  return Number.isNaN(parsed.getTime()) ? '' : dateInputValue(parsed);
}

function isDateInRange(
  value: string | undefined,
  range: { start: string; end: string },
) {
  const key = dateKey(value);
  if (!key) return false;
  return key >= range.start && key <= range.end;
}

function workOrderOverlapsRange(
  workOrder: {
    startDate?: string;
    endDate?: string;
    shifts?: Array<{ date?: string }>;
  },
  range: { start: string; end: string },
) {
  const start = dateKey(workOrder.startDate);
  const end = dateKey(workOrder.endDate || workOrder.startDate);
  if (start && end && start <= range.end && end >= range.start) return true;
  return (workOrder.shifts || []).some((shift) =>
    isDateInRange(shift.date, range),
  );
}

const ReportsView: React.FC = () => {

    workers,
    equipment,
    workOrders,
    timesheets,
    formSubmissions,
    incidents,
    projects,
    settings,
    formTemplates,
    getWorkerById,
    getProjectById,
  } = useAppContext();
  const [reportType, setReportType] = useState('overview');
  const [dateRange, setDateRange] = useState(defaultDateRange);
  const [rangePreset, setRangePreset] = useState('last_30_days');
  const overtimeMultiplier = settings.overtimeRules.overtimeMultiplier || 1;
  const doubleTimeMultiplier = settings.overtimeRules.doubleTimeMultiplier || 1;
  const timesheetPay = (
    timesheet: (typeof timesheets)[number],
    hourlyRate: number,
  ) => {
    const workOrder = workOrders.find(
      (item) => item.id === timesheet.workOrderId,
    );
    const shift = workOrder?.shifts.find(
      (item) => item.id === timesheet.shiftId,
    );
    const start = clockMinutes(
      shift?.defaultRoleStartTime || shift?.startTime || timesheet.clockIn,
    );
    const isNight =
      Boolean(shift?.shiftName?.toLowerCase().includes('night')) ||
      (start !== null && (start >= 17 * 60 || start < 5 * 60));
    const regularRate =
      hourlyRate +
      (isNight ? (settings.overtimeRules.nightShiftPremium ?? 3) : 0);
    const overtimeRate =
      hourlyRate * overtimeMultiplier +
      (isNight ? (settings.overtimeRules.overtimeNightPremium ?? 4.5) : 0);
    const doubleTimeRate =
      hourlyRate * doubleTimeMultiplier +
      (isNight ? (settings.overtimeRules.doubleTimeNightPremium ?? 6) : 0);
    return (
      numericHours(timesheet.regularHours) * regularRate +
      numericHours(timesheet.overtimeHours) * overtimeRate +
      numericHours(timesheet.doubleTimeHours) * doubleTimeRate
    );
  };
  const formTemplateById = new Map(
    formTemplates.map((template) => [template.id, template]),
  );
  const filteredTimesheets = timesheets.filter((timesheet) =>
    isDateInRange(timesheet.date, dateRange),
  );
  const filteredWorkOrders = workOrders.filter((workOrder) =>
    workOrderOverlapsRange(workOrder, dateRange),
  );
  const filteredFormSubmissions = formSubmissions.filter((submission) =>
    isDateInRange(submission.submittedAt, dateRange),
  );
  const filteredIncidents = incidents.filter((incident) =>
    isDateInRange(incident.date, dateRange),
  );
  const activeWorkerIdsInRange = new Set(
    filteredTimesheets.map((timesheet) => timesheet.workerId),
  );

  const activeWorkers = workers.filter(
    (w) => w.status === 'active' && activeWorkerIdsInRange.has(w.id),
  ).length;
  const availableEquip = equipment.filter(
    (e) => e.status === 'available',
  ).length;
  const totalHours = filteredTimesheets.reduce(
    (acc, t) =>
      acc +
      numericHours(t.regularHours) +
      numericHours(t.overtimeHours) +
      numericHours(t.doubleTimeHours),
    0,
  );
  const totalOT = filteredTimesheets.reduce(
    (acc, t) => acc + numericHours(t.overtimeHours),
    0,
  );
  const activeJobs = filteredWorkOrders.length;

  const workerHours = workers
    .filter((w) => w.status === 'active')
    .map((w) => {
      const wTimesheets = filteredTimesheets.filter((t) => t.workerId === w.id);
      const regular = wTimesheets.reduce(
        (a, t) => a + numericHours(t.regularHours),
        0,
      );
      const ot = wTimesheets.reduce(
        (a, t) => a + numericHours(t.overtimeHours),
        0,
      );
      const dt = wTimesheets.reduce(
        (a, t) => a + numericHours(t.doubleTimeHours),
        0,
      );
      const total = regular + ot + dt;
      const pay = wTimesheets.reduce(
        (sum, timesheet) => sum + timesheetPay(timesheet, w.hourlyRate),
        0,
      );
      return { worker: w, regular, ot, total, pay };
    })
    .filter((wh) => wh.total > 0)
    .sort((a, b) => b.total - a.total);
  const maxWorkerHours = Math.max(1, ...workerHours.map((wh) => wh.total));

  const projectSummary = projects
    .map((p) => {
      const pWorkOrders = filteredWorkOrders.filter(
        (wo) => wo.projectId === p.id,
      );
      const pTimesheets = filteredTimesheets.filter(
        (t) => t.projectId === p.id,
      );
      const hours = pTimesheets.reduce(
        (a, t) =>
          a +
          numericHours(t.regularHours) +
          numericHours(t.overtimeHours) +
          numericHours(t.doubleTimeHours),
        0,
      );
      const cost = pTimesheets.reduce((a, t) => {
        const w = getWorkerById(t.workerId);
        return a + (w ? timesheetPay(t, w.hourlyRate) : 0);
      }, 0);
      return { project: p, orders: pWorkOrders.length, hours, cost };
    })

      template,
      count: filteredFormSubmissions.filter(
        (submission) => submission.templateId === template.id,
      ).length,
    }))
    .filter((item) => item.count > 0)
    .sort(
      (a, b) =>
        b.count - a.count || a.template.name.localeCompare(b.template.name),
    );

    const today = new Date();
    let start = new Date(today);
    let end = new Date(today);

    if (preset === 'today') {

    } else if (preset === 'this_week') {
      start = startOfWeek(today);
    } else if (preset === 'last_7_days') {
      start.setDate(today.getDate() - 6);
    } else if (preset === 'this_month') {
      start = startOfMonth(today);
    } else if (preset === 'last_30_days') {
      start.setDate(today.getDate() - 29);
    } else if (preset === 'year_to_date') {
      start = new Date(today.getFullYear(), 0, 1);

      setDateRange({ start: dateInputValue(start), end: dateInputValue(end) });

        'Worker,Type,Regular Hrs,OT Hrs,Total Hrs,Estimated Pay\n' +
        workerHours
          .map(
            (wh) =>
              `${wh.worker.firstName} ${wh.worker.lastName},${wh.worker.type},${wh.regular},${wh.ot},${wh.total},${wh.pay.toFixed(2)}`,

          .join('\n');

        'Project,# Assignments,Total Hours,Labor Cost\n' +
        projectSummary

              `${ps.project.name},${ps.orders},${ps.hours},${ps.cost.toFixed(2)}`,
          )
          .join('\n');

    const a = document.createElement('a');
    a.href = url;
    a.download = `${type}_report.csv`;
    a.click();

            Reports & Analytics
          </h2>
          <p className="text-sm text-gray-500">
            Operational insights and data exports
          </p>
        </div>
        <div className="flex flex-wrap items-center justify-end gap-2">
          <select
            value={rangePreset}

            <option value="this_week">This Week</option>

            <option value="this_month">This Month</option>
            <option value="last_30_days">Last 30 Days</option>
            <option value="year_to_date">Year to Date</option>
            <option value="custom">Custom</option>
          </select>
          <input
            type="date"
            value={dateRange.start}
            onChange={(e) => {

            }}
            className="border border-gray-200 rounded-lg px-3 py-1.5 text-sm"
          />
          <span className="text-gray-400">to</span>
          <input

            value={dateRange.end}
            onChange={(e) => {
              setRangePreset('custom');
              setDateRange((d) => ({ ...d, end: e.target.value }));
            }}
            className="border border-gray-200 rounded-lg px-3 py-1.5 text-sm"
          />
        </div>
      </div>

      {}
      <div className="grid grid-cols-2 md:grid-cols-4 lg:grid-cols-6 gap-4">
        {[
          {
            icon: <Users size={18} className="text-primary" />,
            label: 'Active Workers',
            value: activeWorkers,
            bg: 'bg-blue-100',
          },
          {
            icon: <Truck size={18} className="text-green-600" />,
            label: 'Available Equipment',
            value: availableEquip,
            bg: 'bg-green-100',
          },
          {
            icon: <Briefcase size={18} className="text-purple-600" />,
            label: 'Active Assignments',
            value: activeJobs,
            bg: 'bg-purple-100',
          },
          {
            icon: <Clock size={18} className="text-orange-600" />,
            label: 'Total Hours',
            value: totalHours.toFixed(1),
            bg: 'bg-orange-100',
          },
          {
            icon: <TrendingUp size={18} className="text-red-600" />,
            label: 'Overtime Hours',
            value: totalOT.toFixed(1),
            bg: 'bg-red-100',
          },
          {
            icon: <FileText size={18} className="text-indigo-600" />,
            label: 'Forms Submitted',
            value: filteredFormSubmissions.length,
            bg: 'bg-indigo-100',
          },
        ].map((kpi, i) => (
          <div
            key={i}
            className="bg-white rounded-xl border border-gray-200 p-4"
          >
            <div
              className={`w-8 h-8 ${kpi.bg} rounded-lg flex items-center justify-center mb-2`}
            >
              {kpi.icon}
            </div>
            <p className="text-xl font-bold text-gray-800">{kpi.value}</p>
            <p className="text-xs text-gray-500">{kpi.label}</p>
          </div>
        ))}
      </div>

        {[

          { key: 'worker_hours', label: 'Worker Hours' },
          { key: 'project_summary', label: 'Project Summary' },
        ].map((tab) => (
          <button

            onClick={() => setReportType(tab.key)}
            className={`px-4 py-2 text-sm font-medium border-b-2 -mb-px ${

                ? 'text-primary border-blue-600'
                : 'text-gray-500 border-transparent'
            }`}
          >
            {tab.label}
          </button>
        ))}
      </div>

      {reportType === 'worker_hours' && (
        <div>
          <div className="flex justify-end mb-3">
            <button
              onClick={() => handleExport('worker_hours')}
              className="px-3 py-1.5 border border-gray-300 rounded-lg text-sm text-gray-600 hover:bg-gray-50 flex items-center gap-1"
            >
              <Download size={14} /> Export CSV
            </button>
          </div>
          <div className="bg-white rounded-xl border border-gray-200 overflow-hidden">
            <div
              className="responsive-table"
              role="region"
              aria-label="Table data"
              tabIndex={0}
            >
              <table className="w-full">
                <thead>
                  <tr className="bg-gray-50 border-b border-gray-200">
                    <th className="text-left text-xs font-medium text-gray-500 uppercase px-4 py-3">
                      Worker
                    </th>
                    <th className="text-left text-xs font-medium text-gray-500 uppercase px-4 py-3">
                      Type
                    </th>
                    <th className="text-right text-xs font-medium text-gray-500 uppercase px-4 py-3">
                      Regular
                    </th>
                    <th className="text-right text-xs font-medium text-gray-500 uppercase px-4 py-3">
                      Overtime
                    </th>
                    <th className="text-right text-xs font-medium text-gray-500 uppercase px-4 py-3">
                      Total
                    </th>
                    <th className="text-right text-xs font-medium text-gray-500 uppercase px-4 py-3">
                      Est. Pay
                    </th>
                  </tr>
                </thead>
                <tbody>
                  {workerHours.map((wh) => (
                    <tr
                      key={wh.worker.id}
                      className="border-b border-gray-50 hover:bg-gray-50"
                    >
                      <td className="px-4 py-3 text-sm font-medium text-gray-800">
                        {wh.worker.firstName} {wh.worker.lastName}

                        {wh.worker.type}

                        {wh.regular}h

                      <td className="px-4 py-3 text-sm text-right">

                          className={

                              : 'text-gray-400'

                          {wh.ot}h
                        </span>

                      <td className="px-4 py-3 text-sm font-semibold text-gray-800 text-right">
                        {wh.total}h
                      </td>
                      <td className="px-4 py-3 text-sm font-semibold text-green-700 text-right">
                        ${wh.pay.toFixed(2)}

                    </tr>

                    <td className="px-4 py-3 text-sm" colSpan={2}>
                      Total
                    </td>
                    <td className="px-4 py-3 text-sm text-right">
                      {workerHours.reduce((a, wh) => a + wh.regular, 0)}h

                      {workerHours.reduce((a, wh) => a + wh.ot, 0)}h
                    </td>
                    <td className="px-4 py-3 text-sm text-right">
                      {workerHours.reduce((a, wh) => a + wh.total, 0)}h
                    </td>
                    <td className="px-4 py-3 text-sm text-right text-green-700">
                      ${workerHours.reduce((a, wh) => a + wh.pay, 0).toFixed(2)}

                  </tr>
                </tbody>

            </div>
          </div>
        </div>
      )}

      {reportType === 'project_summary' && (

              Worker Utilization

                  }
                </p>
                <p className="text-sm text-gray-500">Open/Investigating</p>
              </div>
            </div>
            <div className="space-y-2">
              <div className="flex items-center justify-between text-xs font-semibold uppercase text-gray-500 flex-wrap gap-3">
                <span>Submitted Forms</span>
                <span>{filteredFormSubmissions.length}</span>
              </div>
              {formSubmissionSummary.length > 0 ? (
                formSubmissionSummary.map(({ template, count }) => (
                  <div
                    key={template.id}
                    className="flex items-center justify-between rounded-lg bg-gray-50 px-3 py-2 text-sm flex-wrap gap-3"
                  >
                    <div className="min-w-0">
                      <p className="truncate font-medium text-gray-800">
                        {template.name}
                      </p>
                      <p className="truncate text-xs text-gray-500">
                        {template.category || 'Uncategorized'}
                      </p>
                    </div>
                    <span className="ml-3 rounded-full bg-primary/10 px-2 py-0.5 text-xs font-semibold text-primary">
                      {count}
                    </span>
                  </div>
                ))
              ) : (
                <p className="rounded-lg border border-dashed border-gray-200 px-3 py-4 text-center text-sm text-gray-500">
                  No form submissions in this range.
                </p>
              )}
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

export default ReportsView;
