import { usePagedCatalog } from '@/hooks/use-paged-catalog';
import { useDeferredAppData } from '@/hooks/use-deferred-app-data';
import { ON_DEMAND_PAGE_RESOURCES } from '@/lib/appDataPages';
import CatalogPagination from '@/components/shared/CatalogPagination';
import CatalogQueryState from '@/components/shared/CatalogQueryState';
import React, { useState, useMemo } from 'react';
import { useAppContext, AppContextOverride } from '@/contexts/AppContext';
import { useToast } from '@/hooks/use-toast';
import { DEFAULT_ASSIGNMENT_COUNTRY } from '@/lib/googlePlaceAddress';
import PlacesStructuredAddressField from '@/components/shared/PlacesStructuredAddressField';
import CatalogImportButton from '@/components/catalog-import/CatalogImportButton';
import { Project } from '@/lib/types';
import SearchableSelect from '@/components/shared/SearchableSelect';
import {
  Plus,
  Search,
  MapPin,
  Calendar,
  ChevronRight,
  ChevronLeft,
  X,
  Briefcase,
  ExternalLink,
  Edit2,
  FolderKanban,
  Users,
  Clock,
  LayoutList,
  CalendarDays,
  Columns3,
  ArrowUpDown,
  GripVertical,
  Eye,
  Truck,
} from 'lucide-react';

type ViewMode = 'list' | 'calendar' | 'board';
type CalendarMode = 'week' | 'month';

const ProjectsView: React.FC = () => {
  const context = useAppContext();
  const { projectTypes, statusCatalog, setCurrentPage } = context;

  const [search, setSearch] = useState('');
  const [statusFilter, setStatusFilter] = useState('');
  const [selectedProject, setSelectedProject] = useState<string | null>(null);
  const [showNewModal, setShowNewModal] = useState(false);
  const [editingProject, setEditingProject] = useState<string | null>(null);
  const [viewMode, setViewMode] = useState<ViewMode>('list');
  const [calendarMode, setCalendarMode] = useState<CalendarMode>('week');
  const [calendarDate, setCalendarDate] = useState(() => {
    const d = new Date('2026-03-30');
    d.setDate(d.getDate() - d.getDay());
    return d;
  });
  const [sortBy, setSortBy] = useState<'date' | 'name' | 'status'>('date');
  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc');
  const [dragOverStatus, setDragOverStatus] = useState<string | null>(null);
  const [draggedProject, setDraggedProject] = useState<string | null>(null);
  const needsFullProjectViews = viewMode !== 'list' || Boolean(selectedProject);
  const needsRelatedData =
    needsFullProjectViews || Boolean(editingProject || showNewModal);
  const related = useDeferredAppData(
    needsFullProjectViews ? ON_DEMAND_PAGE_RESOURCES.projects! : ['clients'],
    needsRelatedData,
  );
  const catalog = usePagedCatalog<Project>(
    'projects',
    { search, status: statusFilter, sort: sortBy, direction: sortDir },
    viewMode === 'list',
  );
  const projects =
    viewMode === 'list' ? catalog.rows : related.data.projects || [];
  const clients = needsRelatedData
    ? related.data.clients || catalog.relatedClients
    : catalog.relatedClients;
  const workOrders = related.data.workOrders || [];
  const workers = related.data.workers || [];
  const equipment = related.data.equipment || [];
  const getClientById = (id: string) =>
    clients.find((client) => client.id === id);
  const getWorkerById = (id: string) =>
    workers.find((worker) => worker.id === id);
  const { addProject, updateProject } = catalog.overrides;

  const projectStatuses = useMemo(() => {
    const configured = statusCatalog
      .filter((item) => item.scope === 'project' && item.status === 'active')
      .sort(
        (a, b) =>
          (a.sortOrder ?? 0) - (b.sortOrder ?? 0) ||
          (a.name ?? '').localeCompare(b.name ?? ''),
      );
    if (configured.length > 0) return configured;
    return [
      {
        id: 'pj_pending',
        scope: 'project',
        value: 'pending',
        name: 'Pending',
        color: '#F59E0B',
        sortOrder: 10,
        blocksEditing: false,
        triggersNotification: false,
        requiresApproval: false,
        status: 'active',
      },
      {
        id: 'pj_confirmed',
        scope: 'project',
        value: 'confirmed',
        name: 'Confirmed',
        color: '#22C55E',
        sortOrder: 20,
        blocksEditing: false,
        triggersNotification: false,
        requiresApproval: false,
        status: 'active',
      },
      {
        id: 'pj_in_progress',
        scope: 'project',
        value: 'in_progress',
        name: 'In Progress',
        color: '#3B82F6',
        sortOrder: 30,
        blocksEditing: false,
        triggersNotification: false,
        requiresApproval: false,
        status: 'active',
      },
      {
        id: 'pj_completed',
        scope: 'project',
        value: 'completed',
        name: 'Completed',
        color: '#6B7280',
        sortOrder: 40,
        blocksEditing: false,
        triggersNotification: false,
        requiresApproval: false,
        status: 'active',
      },
      {
        id: 'pj_cancelled',
        scope: 'project',
        value: 'cancelled',
        name: 'Cancelled',
        color: '#EF4444',
        sortOrder: 50,
        blocksEditing: true,
        triggersNotification: true,
        requiresApproval: false,
        status: 'active',
      },
    ];
  }, [statusCatalog]);

  const projectStatusMap = useMemo(
    () => new Map(projectStatuses.map((item) => [item.value, item])),
    [projectStatuses],
  );

  const filtered = useMemo(() => {
    if (viewMode === 'list') return projects;
    let result = projects.filter((p) => {
      if (statusFilter && p.status !== statusFilter) return false;
      if (search) {
        const client = getClientById(p.clientId);
        const q = search.toLowerCase();
        return (
          p.name +
          p.number +
          (client?.name || '') +
          p.city +
          p.location +
          p.state +
          p.zipCode
        )
          .toLowerCase()
          .includes(q);
      }
      return true;
    });

    result.sort((a, b) => {
      let cmp = 0;
      if (sortBy === 'date')
        cmp = (a.startDate ?? '').localeCompare(b.startDate ?? '');
      else if (sortBy === 'name')
        cmp = (a.name ?? '').localeCompare(b.name ?? '');
      else if (sortBy === 'status') {
        const order = new Map(
          projectStatuses.map((item, idx) => [item.value, idx]),
        );
        const aOrder = order.get(a.status) ?? Number.MAX_SAFE_INTEGER;
        const bOrder = order.get(b.status) ?? Number.MAX_SAFE_INTEGER;
        cmp = aOrder - bOrder;
      }
      return sortDir === 'asc' ? cmp : -cmp;
    });

    return result;
  }, [
    projects,
    viewMode,
    search,
    statusFilter,
    getClientById,
    sortBy,
    sortDir,
    projectStatuses,
  ]);

  const statusCounts = useMemo(() => {
    const counts: Record<string, number> = { all: projects.length };
    projectStatuses.forEach((status) => {
      counts[status.value] = projects.filter(
        (p) => p.status === status.value,
      ).length;
    });
    return viewMode === 'list'
      ? { all: catalog.meta?.totalItems ?? 0, ...catalog.meta?.statusCounts }
      : counts;
  }, [projects, projectStatuses, viewMode, catalog.meta]);

  const statusColor = (_status: string) => 'text-white border-transparent';

  const statusDot = (_status: string) => '';

  const statusColumnColor = (_status: string) => 'border-gray-300 bg-gray-50';

  const selected = selectedProject
    ? (related.data.projects || projects).find((p) => p.id === selectedProject)
    : null;
  const selectedClient = selected ? getClientById(selected.clientId) : null;
  const selectedProjectType = selected
    ? projectTypes.find((item) => item.id === selected.projectTypeId)
    : null;
  const projectWorkOrders = selected
    ? workOrders.filter((wo) => wo.projectId === selected.id)
    : [];

  const calendarDays = useMemo(() => {
    if (calendarMode === 'week') {
      return Array.from({ length: 7 }, (_, i) => {
        const d = new Date(calendarDate);
        d.setDate(d.getDate() + i);
        return d;
      });
    } else {
      const monthStart = new Date(
        calendarDate.getFullYear(),
        calendarDate.getMonth(),
        1,
      );
      const startDay = monthStart.getDay();
      const daysInMonth = new Date(
        calendarDate.getFullYear(),
        calendarDate.getMonth() + 1,
        0,
      ).getDate();
      const days: Date[] = [];
      for (let i = startDay - 1; i >= 0; i--) {
        const d = new Date(monthStart);
        d.setDate(d.getDate() - i - 1);
        days.push(d);
      }
      for (let i = 1; i <= daysInMonth; i++) {
        days.push(
          new Date(calendarDate.getFullYear(), calendarDate.getMonth(), i),
        );
      }
      while (days.length % 7 !== 0) {
        const last = days[days.length - 1];
        const d = new Date(last);
        d.setDate(d.getDate() + 1);
        days.push(d);
      }
      return days;
    }
  }, [calendarDate, calendarMode]);

  const getProjectsForDate = (date: Date) => {
    const dateStr = date.toISOString().split('T')[0];
    return filtered.filter((p) => {
      const pWOs = workOrders.filter((wo) => wo.projectId === p.id);
      return pWOs.some((wo) => wo.shifts.some((s) => s.date === dateStr));
    });
  };

  const getProjectShiftsForDate = (projectId: string, date: Date) => {
    const dateStr = date.toISOString().split('T')[0];
    const pWOs = workOrders.filter((wo) => wo.projectId === projectId);
    const shifts: { shift: any; wo: any }[] = [];
    pWOs.forEach((wo) => {
      wo.shifts.forEach((s) => {
        if (s.date === dateStr) shifts.push({ shift: s, wo });
      });
    });
    return shifts;
  };

  const navigateCalendar = (dir: number) => {
    setCalendarDate((prev) => {
      const d = new Date(prev);
      if (calendarMode === 'week') d.setDate(d.getDate() + dir * 7);
      else d.setMonth(d.getMonth() + dir);
      return d;
    });
  };

  const goToToday = () => {
    const d = new Date('2026-03-30');
    d.setDate(d.getDate() - d.getDay());
    setCalendarDate(d);
  };

  const toggleSort = (field: 'date' | 'name' | 'status') => {
    if (sortBy === field)
      setSortDir((prev) => (prev === 'asc' ? 'desc' : 'asc'));
    else {
      setSortBy(field);
      setSortDir('asc');
    }
  };

  const handleBoardDragStart = (pId: string) => setDraggedProject(pId);
  const handleBoardDragOver = (e: React.DragEvent, status: string) => {
    e.preventDefault();
    setDragOverStatus(status);
  };
  const handleBoardDrop = (status: string) => {
    if (draggedProject)
      void updateProject(draggedProject, { status: status as any }).catch(
        catalog.reportError,
      );
    setDraggedProject(null);
    setDragOverStatus(null);
  };
  const handleBoardDragEnd = () => {
    setDraggedProject(null);
    setDragOverStatus(null);
  };

  const today = '2026-04-01';
  const boardColumns = projectStatuses.map((item) => item.value);

  if (needsRelatedData && !related.ready)
    return (
      <div className="p-6 space-y-3">
        <CatalogQueryState
          loading={related.loading && !related.error}
          error={related.error}
          retry={related.retry}
        />
        <button
          type="button"
          className="min-h-11 rounded-md border px-3 text-sm"
          onClick={() => {
            setViewMode('list');
            setSelectedProject(null);
            setEditingProject(null);
            setShowNewModal(false);
          }}
        >
          Back to project list
        </button>
      </div>
    );

  return (
    <AppContextOverride
      value={{
        ...related.data,
        ...catalog.overrides,
        projects: needsRelatedData
          ? related.data.projects || projects
          : projects,
        clients: related.data.clients || context.clients,
        getClientById: (id) =>
          related.data.clients?.find((client) => client.id === id) ||
          context.getClientById(id),
        getWorkerById,
      }}
    >
      <div className="min-h-[calc(100dvh-64px)] lg:h-[calc(100dvh-64px)] flex flex-col">
        {}
        <div className="bg-white border-b border-gray-200 px-3 sm:px-6 py-4 flex-shrink-0">
          <div className="flex items-center justify-between mb-3 flex-wrap gap-3">
            <div>
              <h2 className="text-lg font-semibold text-gray-800 flex items-center gap-2">
                <FolderKanban size={22} className="text-primary" />
                Projects
              </h2>
              <p className="text-sm text-gray-500">
                {viewMode === 'list'
                  ? (catalog.meta?.totalItems ?? '—')
                  : projects.length}{' '}
                total projects
              </p>
            </div>
            <div className="flex flex-wrap items-center gap-2">
              {}
              <div className="flex items-center bg-gray-100 rounded-lg p-0.5">
                {[
                  { mode: 'list' as ViewMode, icon: LayoutList, label: 'List' },
                  {
                    mode: 'calendar' as ViewMode,
                    icon: CalendarDays,
                    label: 'Calendar',
                  },
                  { mode: 'board' as ViewMode, icon: Columns3, label: 'Board' },
                ].map((v) => (
                  <button
                    key={v.mode}
                    onClick={() => setViewMode(v.mode)}
                    className={`flex items-center gap-1.5 px-3 py-1.5 rounded-md text-sm font-medium transition-colors ${
                      viewMode === v.mode
                        ? 'bg-white shadow-sm text-primary'
                        : 'text-gray-500 hover:text-gray-700'
                    }`}
                  >
                    <v.icon size={14} />
                    <span className="hidden sm:inline">{v.label}</span>
                  </button>
                ))}
              </div>
              <CatalogImportButton
                scope="projects"
                onDone={() => void catalog.refresh()}
              />
              <button
                disabled={viewMode === 'list' && !catalog.query.data}
                onClick={() => setShowNewModal(true)}
                className="px-4 py-2 bg-primary text-primary-foreground rounded-lg text-sm font-medium hover:bg-primary/90 flex items-center gap-1"
              >
                <Plus size={14} /> New Project
              </button>
            </div>
          </div>

          {}
          <div className="flex items-center gap-3 flex-wrap">
            <div className="flex max-w-full flex-wrap gap-1.5">
              {[
                { key: '', label: 'All', count: statusCounts.all },
                ...projectStatuses.map((status) => ({
                  key: status.value,
                  label: status.name,
                  count: statusCounts[status.value] || 0,
                })),
              ].map((s) => (
                <button
                  key={s.key}
                  onClick={() => setStatusFilter(s.key)}
                  className={`px-3 py-1.5 rounded-lg text-xs font-medium transition-colors ${
                    statusFilter === s.key
                      ? 'bg-primary text-primary-foreground'
                      : 'bg-white border border-gray-200 text-gray-600 hover:bg-gray-50'
                  }`}
                >
                  {s.label} ({s.count})
                </button>
              ))}
            </div>

            <div className="flex items-center bg-white rounded-lg border border-gray-200 px-3 py-1.5 min-w-0 basis-48 basis-48 flex-1 max-w-sm">
              <Search size={14} className="text-gray-400 mr-2" />
              <input
                placeholder="Search projects..."
                value={search}
                onChange={(e) => setSearch(e.target.value)}
                className="bg-transparent text-sm outline-none w-full"
              />
              {search && (
                <button
                  onClick={() => setSearch('')}
                  className="p-0.5 hover:bg-gray-100 rounded"
                >
                  <X size={12} className="text-gray-400" />
                </button>
              )}
            </div>

            {viewMode === 'list' && (
              <div className="flex items-center gap-1">
                {[
                  { key: 'date' as const, label: 'Date' },
                  { key: 'name' as const, label: 'Name' },
                  { key: 'status' as const, label: 'Status' },
                ].map((s) => (
                  <button
                    key={s.key}
                    onClick={() => toggleSort(s.key)}
                    className={`flex items-center gap-1 px-2 py-1 rounded text-xs font-medium transition-colors ${sortBy === s.key ? 'bg-blue-50 text-primary' : 'text-gray-500 hover:bg-gray-100'}`}
                  >
                    <ArrowUpDown size={10} /> {s.label}
                    {sortBy === s.key && (
                      <span className="text-blue-400">
                        {sortDir === 'asc' ? '↑' : '↓'}
                      </span>
                    )}
                  </button>
                ))}
              </div>
            )}

            {viewMode === 'calendar' && (
              <div className="flex items-center gap-2 ml-auto">
                <div className="flex items-center gap-1">
                  <button
                    onClick={() => navigateCalendar(-1)}
                    className="p-1.5 hover:bg-gray-100 rounded-lg"
                  >
                    <ChevronLeft size={16} className="text-gray-500" />
                  </button>
                  <button
                    onClick={goToToday}
                    className="px-3 py-1.5 text-xs text-primary hover:bg-primary/10 rounded-lg font-medium"
                  >
                    Today
                  </button>
                  <button
                    onClick={() => navigateCalendar(1)}
                    className="p-1.5 hover:bg-gray-100 rounded-lg"
                  >
                    <ChevronRight size={16} className="text-gray-500" />
                  </button>
                </div>
                <span className="text-sm font-semibold text-gray-700">
                  {calendarMode === 'week'
                    ? `${calendarDate.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })} - ${new Date(calendarDate.getTime() + 6 * 86400000).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })}`
                    : calendarDate.toLocaleDateString('en-US', {
                        month: 'long',
                        year: 'numeric',
                      })}
                </span>
                <div className="flex items-center bg-gray-100 rounded-lg p-0.5">
                  <button
                    onClick={() => setCalendarMode('week')}
                    className={`px-2.5 py-1 text-xs rounded-md transition-colors ${calendarMode === 'week' ? 'bg-white shadow-sm text-primary font-medium' : 'text-gray-500'}`}
                  >
                    Week
                  </button>
                  <button
                    onClick={() => setCalendarMode('month')}
                    className={`px-2.5 py-1 text-xs rounded-md transition-colors ${calendarMode === 'month' ? 'bg-white shadow-sm text-primary font-medium' : 'text-gray-500'}`}
                  >
                    Month
                  </button>
                </div>
              </div>
            )}
          </div>
        </div>

        {}
        {viewMode === 'list' && (
          <CatalogQueryState
            loading={catalog.query.isPending}
            error={catalog.query.error}
            retry={() => void catalog.query.refetch()}
          />
        )}
        {needsRelatedData && related.ready && related.error && (
          <CatalogQueryState
            loading={false}
            error={related.error}
            retry={related.retry}
          />
        )}
        <div className="flex-1 overflow-auto">
          {}
          {viewMode === 'list' && (
            <div className="p-3 sm:p-6">
              <div className="flex gap-6">
                <div
                  className={`${selectedProject ? 'w-1/2' : 'w-full'} space-y-3 transition-all`}
                >
                  {filtered.map((p) => {
                    const client = getClientById(p.clientId);
                    const woCount =
                      catalog.projectMetrics[p.id]?.workOrders || 0;
                    const totalWorkers =
                      catalog.projectMetrics[p.id]?.workers || 0;

                    return (
                      <div
                        key={p.id}
                        onClick={() => setSelectedProject(p.id)}
                        className={`bg-white rounded-xl border p-4 cursor-pointer transition-all ${
                          selectedProject === p.id
                            ? 'border-blue-400 shadow-md'
                            : 'border-gray-200 hover:shadow-md'
                        }`}
                      >
                        <div className="flex items-start justify-between mb-2">
                          <div>
                            <div className="flex items-center gap-2">
                              <span className="text-xs font-mono text-gray-400 bg-gray-50 px-1.5 py-0.5 rounded">
                                #{p.number}
                              </span>
                              <h3 className="font-semibold text-gray-800">
                                {p.name}
                              </h3>
                            </div>
                            <p className="text-sm text-gray-500 mt-0.5">
                              {client?.name || 'No client'}
                            </p>
                          </div>
                          <span
                            className={`text-xs px-2 py-1 rounded-full font-medium capitalize border ${statusColor(p.status)}`}
                            style={{
                              backgroundColor:
                                projectStatusMap.get(p.status)?.color ||
                                undefined,
                            }}
                          >
                            {projectStatusMap.get(p.status)?.name ||
                              p.status.replace('_', ' ')}
                          </span>
                        </div>
                        <div className="flex items-center gap-4 text-xs text-gray-500">
                          <span className="flex items-center gap-1">
                            <MapPin size={12} /> {p.city}
                          </span>
                          <span className="flex items-center gap-1">
                            <Calendar size={12} /> {p.startDate} - {p.endDate}
                          </span>
                          <span className="flex items-center gap-1">
                            <Briefcase size={12} /> {woCount} orders
                          </span>
                          <span className="flex items-center gap-1">
                            <Users size={12} /> {totalWorkers} workers
                          </span>
                        </div>
                      </div>
                    );
                  })}
                  {filtered.length === 0 && (
                    <div className="text-center py-12 text-gray-400">
                      <FolderKanban
                        size={40}
                        className="mx-auto mb-2 opacity-50"
                      />
                      <p>No projects found</p>
                    </div>
                  )}
                </div>

                {}
                {selected && (
                  <div className="w-1/2 bg-white rounded-xl border border-gray-200 sticky top-4 self-start max-h-[calc(100vh-200px)] overflow-y-auto">
                    <div className="p-5 border-b border-gray-200 flex items-center justify-between flex-wrap gap-3">
                      <div>
                        <div className="flex items-center gap-2">
                          <span className="text-xs font-mono text-gray-400 bg-gray-50 px-1.5 py-0.5 rounded">
                            #{selected.number}
                          </span>
                          <h3 className="font-semibold text-gray-800">
                            {selected.name}
                          </h3>
                        </div>
                      </div>
                      <div className="flex items-center gap-2">
                        <button
                          onClick={() => setEditingProject(selected.id)}
                          className="p-1.5 hover:bg-gray-100 rounded-lg"
                          title="Edit"
                        >
                          <Edit2 size={16} className="text-gray-400" />
                        </button>
                        <button
                          onClick={() => setCurrentPage('scheduler')}
                          className="p-1.5 hover:bg-primary/10 rounded-lg text-primary"
                          title="Open in Schedule"
                        >
                          <CalendarDays size={16} />
                        </button>
                        <button onClick={() => setSelectedProject(null)}>
                          <X size={18} className="text-gray-400" />
                        </button>
                      </div>
                    </div>
                    <div className="p-5 space-y-4">
                      <div className="responsive-grid grid grid-cols-2 gap-3 text-sm">
                        <div>
                          <span className="text-gray-500">Client:</span>
                          <p className="font-medium">{selectedClient?.name}</p>
                        </div>
                        <div>
                          <span className="text-gray-500">Project Type:</span>
                          <p className="font-medium">
                            {selectedProjectType?.name || '—'}
                          </p>
                        </div>
                        <div>
                          <span className="text-gray-500">Project #:</span>
                          <p className="font-medium">{selected.number}</p>
                        </div>
                        <div>
                          <span className="text-gray-500">Location:</span>
                          <p className="font-medium">
                            {selected.location}, {selected.city}
                          </p>
                        </div>
                        <div>
                          <span className="text-gray-500">
                            Project Manager:
                          </span>
                          <p className="font-medium">
                            {selected.projectManager || '—'}
                          </p>
                        </div>
                        <div>
                          <span className="text-gray-500">Manager Email:</span>
                          <p className="font-medium">
                            {selected.projectManagerEmail || '—'}
                          </p>
                        </div>
                        <div>
                          <span className="text-gray-500">Status:</span>
                          <select
                            value={selected.status}
                            onChange={(e) =>
                              updateProject(selected.id, {
                                status: e.target.value as any,
                              })
                            }
                            className="block text-xs px-2 py-1 rounded-full font-medium mt-0.5 border border-gray-200"
                          >
                            {projectStatuses.map((status) => (
                              <option key={status.id} value={status.value}>
                                {status.name}
                              </option>
                            ))}
                          </select>
                        </div>
                        <div>
                          <span className="text-gray-500">Start:</span>
                          <p className="font-medium">{selected.startDate}</p>
                        </div>
                        <div>
                          <span className="text-gray-500">End:</span>
                          <p className="font-medium">{selected.endDate}</p>
                        </div>
                      </div>

                      {selected.description && (
                        <div>
                          <span className="text-sm text-gray-500">
                            Description:
                          </span>
                          <p className="text-sm text-gray-700 mt-1 bg-gray-50 rounded p-2">
                            {selected.description}
                          </p>
                        </div>
                      )}
                      {selected.notes && (
                        <div>
                          <span className="text-sm text-gray-500">Notes:</span>
                          <p className="text-sm text-gray-700 mt-1 bg-gray-50 rounded p-2">
                            {selected.notes}
                          </p>
                        </div>
                      )}

                      {}
                      <div>
                        <div className="flex items-center justify-between mb-2 flex-wrap gap-3">
                          <h4 className="font-semibold text-gray-700">
                            Assignments ({projectWorkOrders.length})
                          </h4>
                          <button
                            onClick={() => setCurrentPage('scheduler')}
                            className="text-xs text-primary hover:text-primary/90 flex items-center gap-1"
                          >
                            <Plus size={12} /> Add Assignment
                          </button>
                        </div>
                        {projectWorkOrders.length > 0 ? (
                          <div className="space-y-2">
                            {projectWorkOrders.map((wo) => {
                              const totalAssigned = wo.shifts.reduce(
                                (acc, s) =>
                                  acc +
                                  s.roles.reduce(
                                    (a, r) => a + r.assignedWorkers.length,
                                    0,
                                  ),
                                0,
                              );
                              const totalRequired = wo.shifts.reduce(
                                (acc, s) =>
                                  acc +
                                  s.roles.reduce(
                                    (a, r) => a + r.requiredCount,
                                    0,
                                  ),
                                0,
                              );
                              return (
                                <div
                                  key={wo.id}
                                  className="border border-gray-200 rounded-lg p-3 hover:bg-gray-50 cursor-pointer"
                                  onClick={() => setCurrentPage('workorders')}
                                >
                                  <div className="flex items-center justify-between mb-1 flex-wrap gap-3">
                                    <div className="flex items-center gap-2">
                                      <span className="text-sm font-medium text-gray-800">
                                        {wo.title}
                                      </span>
                                      {wo.orderNumber && (
                                        <span className="text-xs font-mono text-gray-400 bg-gray-100 px-1.5 py-0.5 rounded">
                                          {wo.orderNumber}
                                        </span>
                                      )}
                                    </div>
                                    <ExternalLink
                                      size={14}
                                      className="text-gray-400"
                                    />
                                  </div>
                                  <div className="flex items-center gap-3 text-xs text-gray-500">
                                    <span>
                                      {wo.startDate} - {wo.endDate}
                                    </span>
                                    <span
                                      className={
                                        totalAssigned < totalRequired
                                          ? 'text-red-600 font-medium'
                                          : 'text-green-600 font-medium'
                                      }
                                    >
                                      {totalAssigned}/{totalRequired} workers
                                    </span>
                                  </div>
                                </div>
                              );
                            })}
                          </div>
                        ) : (
                          <p className="text-sm text-gray-400 italic">
                            No assignments yet
                          </p>
                        )}
                      </div>

                      {}
                      {projectWorkOrders.length > 0 &&
                        (() => {
                          const workerIds = new Set<string>();
                          projectWorkOrders.forEach((wo) =>
                            wo.shifts.forEach((s) =>
                              s.roles.forEach((r) =>
                                r.assignedWorkers.forEach((wid) =>
                                  workerIds.add(wid),
                                ),
                              ),
                            ),
                          );
                          const uniqueWorkers = Array.from(workerIds)
                            .map((wid) => getWorkerById(wid))
                            .filter(Boolean);
                          if (uniqueWorkers.length === 0) return null;
                          return (
                            <div>
                              <h4 className="font-semibold text-gray-700 mb-2">
                                Assigned Workers ({uniqueWorkers.length})
                              </h4>
                              <div className="flex flex-wrap gap-2">
                                {uniqueWorkers.slice(0, 8).map(
                                  (w) =>
                                    w && (
                                      <div
                                        key={w.id}
                                        className="flex items-center gap-1.5 bg-gray-50 rounded-lg px-2 py-1 border border-gray-100"
                                      >
                                        <div className="w-6 h-6 rounded-full bg-blue-500 flex items-center justify-center text-white text-xs font-bold">
                                          {w.firstName[0]}
                                          {w.lastName[0]}
                                        </div>
                                        <span className="text-xs font-medium text-gray-700">
                                          {w.firstName} {w.lastName}
                                        </span>
                                        <span className="text-xs text-gray-400">
                                          {w.type}
                                        </span>
                                      </div>
                                    ),
                                )}
                                {uniqueWorkers.length > 8 && (
                                  <span className="text-xs text-gray-400 self-center">
                                    +{uniqueWorkers.length - 8} more
                                  </span>
                                )}
                              </div>
                            </div>
                          );
                        })()}
                    </div>
                  </div>
                )}
              </div>
            </div>
          )}

          {}
          {viewMode === 'calendar' && (
            <div className="p-4">
              {calendarMode === 'week' ? (
                <div className="bg-white rounded-xl border border-gray-200 overflow-hidden">
                  <div className="grid grid-cols-7 border-b border-gray-200">
                    {calendarDays.map((d, i) => {
                      const dateStr = d.toISOString().split('T')[0];
                      const isToday = dateStr === today;
                      const isWeekend = d.getDay() === 0 || d.getDay() === 6;
                      return (
                        <div
                          key={i}
                          className={`text-center py-3 border-r border-gray-100 last:border-r-0 ${isToday ? 'bg-blue-50' : isWeekend ? 'bg-gray-50' : ''}`}
                        >
                          <p
                            className={`text-xs font-semibold ${isToday ? 'text-primary' : isWeekend ? 'text-gray-400' : 'text-gray-600'}`}
                          >
                            {d.toLocaleDateString('en-US', {
                              weekday: 'short',
                            })}
                          </p>
                          <p
                            className={`text-lg font-bold ${isToday ? 'text-primary' : 'text-gray-800'}`}
                          >
                            {d.getDate()}
                          </p>
                          <p className="text-xs text-gray-400">
                            {d.toLocaleDateString('en-US', { month: 'short' })}
                          </p>
                        </div>
                      );
                    })}
                  </div>
                  <div className="grid grid-cols-7 min-h-[500px]">
                    {calendarDays.map((d, i) => {
                      const dateStr = d.toISOString().split('T')[0];
                      const isToday = dateStr === today;
                      const isWeekend = d.getDay() === 0 || d.getDay() === 6;
                      const dayProjects = getProjectsForDate(d);

                      return (
                        <div
                          key={i}
                          className={`border-r border-gray-100 last:border-r-0 p-2 space-y-1.5 ${isToday ? 'bg-blue-50/30' : isWeekend ? 'bg-gray-50/50' : ''}`}
                        >
                          {dayProjects.map((p) => {
                            const client = getClientById(p.clientId);
                            const shifts = getProjectShiftsForDate(p.id, d);
                            const totalWorkers = new Set<string>();
                            shifts.forEach(({ shift }) =>
                              shift.roles.forEach((r: any) =>
                                r.assignedWorkers.forEach((wid: string) =>
                                  totalWorkers.add(wid),
                                ),
                              ),
                            );

                            return (
                              <button
                                key={p.id}
                                onClick={() => setSelectedProject(p.id)}
                                className={`w-full text-left rounded-lg border p-2 transition-all hover:shadow-sm ${
                                  selectedProject === p.id
                                    ? 'border-blue-400 bg-blue-50 shadow-sm'
                                    : 'border-gray-200 bg-white hover:border-gray-300'
                                }`}
                              >
                                <div className="flex items-center gap-1.5 mb-1">
                                  <div
                                    className={`w-2 h-2 rounded-full flex-shrink-0 ${statusDot(p.status)}`}
                                    style={{
                                      backgroundColor:
                                        projectStatusMap.get(p.status)?.color ||
                                        undefined,
                                    }}
                                  />
                                  <span className="text-xs font-semibold text-gray-800 truncate">
                                    {p.name}
                                  </span>
                                </div>
                                <div className="text-xs text-gray-500 truncate">
                                  {client?.name}
                                </div>
                                <div className="flex items-center gap-2 mt-1 text-xs text-gray-400">
                                  <span className="flex items-center gap-0.5">
                                    <Briefcase size={10} /> {shifts.length}
                                  </span>
                                  <span className="flex items-center gap-0.5">
                                    <Users size={10} /> {totalWorkers.size}
                                  </span>
                                </div>
                              </button>
                            );
                          })}
                          {dayProjects.length === 0 && (
                            <div className="h-full flex items-center justify-center">
                              <span className="text-xs text-gray-300">-</span>
                            </div>
                          )}
                        </div>
                      );
                    })}
                  </div>
                </div>
              ) : (
                <div className="bg-white rounded-xl border border-gray-200 overflow-hidden">
                  <div className="grid grid-cols-7 border-b border-gray-200">
                    {['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'].map(
                      (d) => (
                        <div
                          key={d}
                          className="text-center py-2 text-xs font-semibold text-gray-500 border-r border-gray-100 last:border-r-0"
                        >
                          {d}
                        </div>
                      ),
                    )}
                  </div>
                  <div className="grid grid-cols-7">
                    {calendarDays.map((d, i) => {
                      const dateStr = d.toISOString().split('T')[0];
                      const isToday = dateStr === today;
                      const isCurrentMonth =
                        d.getMonth() === calendarDate.getMonth();
                      const dayProjects = getProjectsForDate(d);

                      return (
                        <div
                          key={i}
                          className={`min-h-[100px] border-r border-b border-gray-100 p-1.5 ${isToday ? 'bg-blue-50/50' : !isCurrentMonth ? 'bg-gray-50/50 opacity-50' : ''}`}
                        >
                          <div
                            className={`text-xs font-bold mb-1 ${isToday ? 'text-primary' : 'text-gray-700'}`}
                          >
                            {d.getDate()}
                          </div>
                          <div className="space-y-0.5">
                            {dayProjects.slice(0, 3).map((p) => (
                              <button
                                key={p.id}
                                onClick={() => setSelectedProject(p.id)}
                                className="w-full text-left text-xs px-1.5 py-0.5 rounded truncate hover:opacity-80"
                                style={{
                                  backgroundColor:
                                    projectStatusMap.get(p.status)?.color ||
                                    '#e5e7eb',
                                }}
                              >
                                {p.name}
                              </button>
                            ))}
                            {dayProjects.length > 3 && (
                              <span className="text-xs text-gray-400 px-1">
                                +{dayProjects.length - 3}
                              </span>
                            )}
                          </div>
                        </div>
                      );
                    })}
                  </div>
                </div>
              )}

              {}
              {selected && (
                <div className="mt-4 bg-white rounded-xl border border-gray-200 p-5">
                  <div className="flex items-center justify-between mb-3 flex-wrap gap-3">
                    <div className="flex items-center gap-2">
                      <FolderKanban size={16} className="text-primary" />
                      <span className="text-xs font-mono text-gray-400">
                        #{selected.number}
                      </span>
                      <h3 className="font-semibold text-gray-800">
                        {selected.name}
                      </h3>
                      <span
                        className={`text-xs px-2 py-0.5 rounded-full font-medium capitalize border ${statusColor(selected.status)}`}
                        style={{
                          backgroundColor:
                            projectStatusMap.get(selected.status)?.color ||
                            undefined,
                        }}
                      >
                        {projectStatusMap.get(selected.status)?.name ||
                          selected.status.replace('_', ' ')}
                      </span>
                    </div>
                    <div className="flex items-center gap-2">
                      <button
                        onClick={() => setCurrentPage('scheduler')}
                        className="text-xs text-primary hover:text-primary/90 flex items-center gap-1 px-2 py-1 bg-blue-50 rounded-lg"
                      >
                        <Eye size={12} /> Schedule
                      </button>
                      <button onClick={() => setSelectedProject(null)}>
                        <X size={18} className="text-gray-400" />
                      </button>
                    </div>
                  </div>
                  <div className="text-sm text-gray-500 mb-3">
                    {selectedClient?.name} | {selected.location},{' '}
                    {selected.city} | {selected.startDate} - {selected.endDate}
                  </div>
                  <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
                    {projectWorkOrders.map((wo) => (
                      <div
                        key={wo.id}
                        className="border border-gray-200 rounded-lg p-2.5"
                      >
                        <div className="flex items-center gap-1.5 mb-1">
                          <span className="text-xs font-medium text-gray-800 truncate">
                            {wo.title}
                          </span>
                        </div>
                        {wo.orderNumber && (
                          <span className="text-xs font-mono text-gray-400">
                            {wo.orderNumber}
                          </span>
                        )}
                        <div className="text-xs text-gray-500 mt-1">
                          {wo.shifts.length} shifts | {wo.startDate} -{' '}
                          {wo.endDate}
                        </div>
                      </div>
                    ))}
                  </div>
                </div>
              )}
            </div>
          )}

          {}
          {viewMode === 'board' && (
            <div className="p-4 h-full">
              <div className="flex gap-4 h-full overflow-x-auto pb-4">
                {boardColumns.map((status) => {
                  const columnProjects = filtered.filter(
                    (p) => p.status === status,
                  );
                  const isDragOver = dragOverStatus === status;
                  return (
                    <div
                      key={status}
                      className={`flex-shrink-0 w-72 flex flex-col rounded-xl border-2 transition-colors ${isDragOver ? 'border-blue-400 bg-blue-50/50' : statusColumnColor(status)}`}
                      onDragOver={(e) => handleBoardDragOver(e, status)}
                      onDragLeave={() => setDragOverStatus(null)}
                      onDrop={() => handleBoardDrop(status)}
                    >
                      <div className="px-3 py-3 border-b border-gray-200/50 flex items-center justify-between flex-wrap gap-3">
                        <div className="flex items-center gap-2">
                          <div
                            className={`w-3 h-3 rounded-full ${statusDot(status)}`}
                            style={{
                              backgroundColor:
                                projectStatusMap.get(status)?.color ||
                                undefined,
                            }}
                          />
                          <span className="text-sm font-semibold text-gray-800 capitalize">
                            {projectStatusMap.get(status)?.name ||
                              status.replace('_', ' ')}
                          </span>
                        </div>
                        <span className="text-xs font-bold text-gray-500 bg-white/80 px-2 py-0.5 rounded-full">
                          {columnProjects.length}
                        </span>
                      </div>
                      <div className="flex-1 overflow-y-auto p-2 space-y-2">
                        {columnProjects.map((p) => {
                          const client = getClientById(p.clientId);
                          const woCount = workOrders.filter(
                            (wo) => wo.projectId === p.id,
                          ).length;
                          const isDragging = draggedProject === p.id;

                          return (
                            <div
                              key={p.id}
                              draggable
                              onDragStart={() => handleBoardDragStart(p.id)}
                              onDragEnd={handleBoardDragEnd}
                              onClick={() => setSelectedProject(p.id)}
                              className={`bg-white rounded-lg border border-gray-200 p-3 cursor-grab active:cursor-grabbing transition-all hover:shadow-md ${isDragging ? 'opacity-50 scale-95' : ''} ${selectedProject === p.id ? 'ring-2 ring-blue-400' : ''}`}
                            >
                              <div className="flex items-start justify-between mb-1.5">
                                <div className="min-w-0 flex-1">
                                  <div className="flex items-center gap-1.5 mb-0.5">
                                    <span className="text-xs font-mono text-gray-400">
                                      #{p.number}
                                    </span>
                                  </div>
                                  <p className="text-sm font-semibold text-gray-800 truncate">
                                    {p.name}
                                  </p>
                                  <p className="text-xs text-gray-500 truncate">
                                    {client?.name}
                                  </p>
                                </div>
                                <GripVertical
                                  size={14}
                                  className="text-gray-300 flex-shrink-0 mt-0.5"
                                />
                              </div>
                              <div className="flex items-center gap-3 text-xs text-gray-500 mt-2">
                                <span className="flex items-center gap-1">
                                  <MapPin size={10} /> {p.city}
                                </span>
                                <span className="flex items-center gap-1">
                                  <Briefcase size={10} /> {woCount}
                                </span>
                              </div>
                              <div className="mt-1.5 text-xs text-gray-400">
                                {p.startDate} - {p.endDate}
                              </div>
                            </div>
                          );
                        })}
                        {columnProjects.length === 0 && (
                          <div className="text-center py-8 text-gray-400">
                            <p className="text-xs">No projects</p>
                          </div>
                        )}
                      </div>
                    </div>
                  );
                })}
              </div>
            </div>
          )}
        </div>

        {}
        {showNewModal && (
          <NewProjectModal onClose={() => setShowNewModal(false)} />
        )}
        {editingProject && (
          <EditProjectModal
            projectId={editingProject}
            onClose={() => setEditingProject(null)}
          />
        )}
        {viewMode === 'list' && (
          <CatalogPagination
            meta={catalog.meta}
            page={catalog.page}
            onPage={catalog.setPage}
            busy={catalog.query.isFetching}
          />
        )}
      </div>
    </AppContextOverride>
  );
};

const NewProjectModal: React.FC<{ onClose: () => void }> = ({ onClose }) => {
  const { clients, projectTypes, statusCatalog, addProject } = useAppContext();
  const { toast } = useToast();
  const projectStatuses = useMemo(() => {
    const configured = statusCatalog
      .filter((item) => item.scope === 'project' && item.status === 'active')
      .sort(
        (a, b) =>
          (a.sortOrder ?? 0) - (b.sortOrder ?? 0) ||
          (a.name ?? '').localeCompare(b.name ?? ''),
      );
    if (configured.length > 0) return configured;
    return [
      {
        id: 'pj_pending',
        scope: 'project',
        value: 'pending',
        name: 'Pending',
        color: '#F59E0B',
        sortOrder: 10,
        blocksEditing: false,
        triggersNotification: false,
        requiresApproval: false,
        status: 'active',
      },
    ];
  }, [statusCatalog]);
  const activeProjectTypes = useMemo(
    () =>
      projectTypes
        .filter((item) => item.status === 'active')
        .sort((a, b) => (a.name ?? '').localeCompare(b.name ?? '')),
    [projectTypes],
  );
  const clientOptions = useMemo(
    () =>
      clients.map((client) => ({
        value: client.id,
        label: client.name,
        searchText: `${client.contactName} ${client.email} ${client.website || ''}`,
      })),
    [clients],
  );
  const [form, setForm] = useState({
    number: '',
    name: '',
    clientId: '',
    projectTypeId: activeProjectTypes[0]?.id || '',
    projectManager: '',
    projectManagerEmail: '',
    location: '',
    city: '',
    state: '',
    zipCode: '',
    country: DEFAULT_ASSIGNMENT_COUNTRY,
    latitude: null as number | null,
    longitude: null as number | null,
    status: projectStatuses[0]?.value || 'pending',
    startDate: '2026-04-07',
    endDate: '2026-04-18',
    description: '',
    notes: '',
  });

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!form.clientId) {
      toast({
        title: 'Missing client',
        description: 'Please select a client.',
        variant: 'destructive',
      });
      return;
    }
    addProject({ id: `p_new_${Date.now()}`, ...form });
    onClose();
  };

  return (
    <div className="fixed inset-0 bg-black/50 z-50 flex items-center justify-center p-2 sm:p-4">
      <div
        className="bg-white rounded-xl w-full max-w-xl max-h-[90dvh] overflow-y-auto"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="p-5 border-b border-gray-200 flex items-center justify-between flex-wrap gap-3">
          <h2 className="text-lg font-semibold text-gray-800">New Project</h2>
          <button
            onClick={onClose}
            className="p-1 hover:bg-gray-100 rounded-lg"
          >
            <X size={18} className="text-gray-400" />
          </button>
        </div>
        <form onSubmit={handleSubmit} className="p-5 space-y-4">
          <div className="responsive-grid grid grid-cols-2 gap-3">
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">
                Project # *
              </label>
              <input
                required
                value={form.number}
                onChange={(e) =>
                  setForm((f) => ({ ...f, number: e.target.value }))
                }
                className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
                placeholder="e.g., 1700"
              />
            </div>
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">
                Status
              </label>
              <select
                value={form.status}
                onChange={(e) =>
                  setForm((f) => ({ ...f, status: e.target.value as any }))
                }
                className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
              >
                {projectStatuses.map((status) => (
                  <option key={status.id} value={status.value}>
                    {status.name}
                  </option>
                ))}
              </select>
            </div>
          </div>
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">
              Project Name *
            </label>
            <input
              required
              value={form.name}
              onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
              placeholder="Project name"
            />
          </div>
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">
              Client *
            </label>
            <SearchableSelect
              value={form.clientId}
              onChange={(clientId) => setForm((f) => ({ ...f, clientId }))}
              options={clientOptions}
              placeholder="Select client..."
              searchPlaceholder="Search client..."
              emptyMessage="No clients found."
            />
          </div>
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">
              Project Type
            </label>
            <select
              value={form.projectTypeId}
              onChange={(e) =>
                setForm((f) => ({ ...f, projectTypeId: e.target.value }))
              }
              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
            >
              <option value="">Select project type...</option>
              {activeProjectTypes.map((type) => (
                <option key={type.id} value={type.id}>
                  {type.name}
                </option>
              ))}
            </select>
          </div>
          <div className="responsive-grid grid grid-cols-2 gap-3">
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">
                Project Manager
              </label>
              <input
                value={form.projectManager}
                onChange={(e) =>
                  setForm((f) => ({ ...f, projectManager: e.target.value }))
                }
                className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
                placeholder="Manager name"

const EditProjectModal: React.FC<{
  projectId: string;
  onClose: () => void;
}> = ({ projectId, onClose }) => {
  const { projects, clients, projectTypes, statusCatalog, updateProject } =
    useAppContext();
  const { toast } = useToast();
  const projectStatuses = useMemo(() => {
    const configured = statusCatalog
      .filter((item) => item.scope === 'project' && item.status === 'active')
      .sort(
        (a, b) =>
          (a.sortOrder ?? 0) - (b.sortOrder ?? 0) ||
          (a.name ?? '').localeCompare(b.name ?? ''),
      );
    if (configured.length > 0) return configured;
    return [
      {
        id: 'pj_pending',
        scope: 'project',
        value: 'pending',
        name: 'Pending',
        color: '#F59E0B',
        sortOrder: 10,
        blocksEditing: false,
        triggersNotification: false,
        requiresApproval: false,
        status: 'active',
      },
    ];
  }, [statusCatalog]);
  const activeProjectTypes = useMemo(
    () =>
      projectTypes
        .filter((item) => item.status === 'active')
        .sort((a, b) => (a.name ?? '').localeCompare(b.name ?? '')),
    [projectTypes],
  );
  const clientOptions = useMemo(
    () =>
      clients.map((client) => ({
        value: client.id,
        label: client.name,
        searchText: `${client.contactName} ${client.email} ${client.website || ''}`,
      })),
    [clients],
  );
  const project = projects.find((p) => p.id === projectId);
  const [form, setForm] = useState({
    number: project?.number || '',
    name: project?.name || '',
    clientId: project?.clientId || '',
    projectTypeId: project?.projectTypeId || '',
    projectManager: project?.projectManager || '',
    projectManagerEmail: project?.projectManagerEmail || '',
    location: project?.location || '',
    city: project?.city || '',
    state: project?.state || '',
    zipCode: project?.zipCode || '',
    country: project?.country || DEFAULT_ASSIGNMENT_COUNTRY,
    latitude:
      project?.latitude !== undefined && project?.latitude !== null
        ? Number(project.latitude)
        : null,
    longitude:
      project?.longitude !== undefined && project?.longitude !== null
        ? Number(project.longitude)
        : null,
    status: project?.status || 'pending',
    startDate: project?.startDate || '',
    endDate: project?.endDate || '',
    description: project?.description || '',
    notes: project?.notes || '',
  });
  const [saving, setSaving] = useState(false);

  if (!project) return null;

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!form.clientId) {
      toast({
        title: 'Missing client',
        description: 'Please select a client.',
        variant: 'destructive',
      });
      return;
    }
    setSaving(true);
    try {
      await updateProject(projectId, {
        ...form,
        startDate: form.startDate || null,
        endDate: form.endDate || null,
      } as any);
      toast({
        title: 'Project updated',
        description: 'The project changes were saved successfully.',
      });
      onClose();
    } catch (error) {
      toast({
        title: 'Could not update project',
        description:
          error instanceof Error
            ? error.message
            : 'Please review the project information and try again.',
        variant: 'destructive',
      });
    } finally {
      setSaving(false);
    }
  };

  return (
    <div className="fixed inset-0 bg-black/50 z-50 flex items-center justify-center p-2 sm:p-4">
      <div
        className="bg-white rounded-xl w-full max-w-xl max-h-[90dvh] overflow-y-auto"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="p-5 border-b border-gray-200 flex items-center justify-between flex-wrap gap-3">
          <h2 className="text-lg font-semibold text-gray-800">Edit Project</h2>
          <button
            onClick={onClose}
            className="p-1 hover:bg-gray-100 rounded-lg"
          >
            <X size={18} className="text-gray-400" />
          </button>
        </div>
        <form onSubmit={handleSubmit} className="p-5 space-y-4">
          <div className="responsive-grid grid grid-cols-2 gap-3">
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">
                Project #
              </label>
              <input
                value={form.number}
                onChange={(e) =>
                  setForm((f) => ({ ...f, number: e.target.value }))
                }
                className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
              />
            </div>
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">
                Status
              </label>
              <select
                value={form.status}
                onChange={(e) =>
                  setForm((f) => ({ ...f, status: e.target.value as any }))
                }
                className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
              >
                {projectStatuses.map((status) => (
                  <option key={status.id} value={status.value}>
                    {status.name}
                  </option>
                ))}
              </select>
            </div>
          </div>
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">
              Project Name
            </label>
            <input
              value={form.name}
              onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
            />
          </div>
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">
              Client
            </label>
            <SearchableSelect
              value={form.clientId}
              onChange={(clientId) => setForm((f) => ({ ...f, clientId }))}
              options={clientOptions}
              placeholder="Select client..."
              searchPlaceholder="Search client..."
              emptyMessage="No clients found."
            />
          </div>
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">
              Project Type
            </label>
            <select
              value={form.projectTypeId}
              onChange={(e) =>
                setForm((f) => ({ ...f, projectTypeId: e.target.value }))
              }
              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
            >
              <option value="">Select project type...</option>
              {activeProjectTypes.map((type) => (
                <option key={type.id} value={type.id}>
                  {type.name}
                </option>
              ))}
            </select>
          </div>
          <div className="responsive-grid grid grid-cols-2 gap-3">
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">
                Project Manager
              </label>
              <input
                value={form.projectManager}
                onChange={(e) =>
                  setForm((f) => ({ ...f, projectManager: e.target.value }))
                }
                className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
              />
            </div>
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">
                Manager Email
              </label>
              <input
                type="email"
                value={form.projectManagerEmail}
                onChange={(e) =>
                  setForm((f) => ({
                    ...f,
                    projectManagerEmail: e.target.value,
                  }))
                }
                className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
              />
            </div>
          </div>
          <PlacesStructuredAddressField
            id={`project-location-${projectId}`}
            label="Job site address"
            value={{
              addressLine: form.location,
              latitude: form.latitude ?? null,
              longitude: form.longitude ?? null,
              city: form.city,
              state: form.state,
              zipCode: form.zipCode,
              country: form.country || DEFAULT_ASSIGNMENT_COUNTRY,
            }}
            onChange={(next) =>
              setForm((f) => ({
                ...f,
                location: next.addressLine,
                latitude: next.latitude,
                longitude: next.longitude,
                city: next.city,
                state: next.state,
                zipCode: next.zipCode,
                country: next.country || DEFAULT_ASSIGNMENT_COUNTRY,
              }))
            }
          />
          <div className="responsive-grid grid grid-cols-2 gap-3">
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">
                Start Date
              </label>
              <input
                type="date"
                value={form.startDate}
                onChange={(e) =>
                  setForm((f) => ({ ...f, startDate: e.target.value }))
                }
                className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
              />
            </div>
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-1">
                End Date
              </label>
              <input
                type="date"
                value={form.endDate}
                onChange={(e) =>
                  setForm((f) => ({ ...f, endDate: e.target.value }))
                }
                className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
              />
            </div>
          </div>
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">
              Description
            </label>
            <textarea
              value={form.description}
              onChange={(e) =>
                setForm((f) => ({ ...f, description: e.target.value }))
              }
              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
              rows={2}
            />
          </div>
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-1">
              Notes
            </label>
            <textarea
              value={form.notes}
              onChange={(e) =>
                setForm((f) => ({ ...f, notes: e.target.value }))
              }
              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm"
              rows={2}
            />
          </div>
          <div className="flex gap-3 pt-2">
            <button
              type="button"
              disabled={saving}
              onClick={onClose}
              className="flex-1 px-4 py-2 border border-gray-300 rounded-lg text-sm font-medium text-gray-700 hover:bg-gray-50 disabled:opacity-50"
            >
              Cancel
            </button>
            <button
              type="submit"
              disabled={saving}
              className="flex-1 px-4 py-2 bg-primary text-primary-foreground rounded-lg text-sm font-medium hover:bg-primary/90 disabled:opacity-50"
            >
              {saving ? 'Saving...' : 'Save Changes'}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
};

export default ProjectsView;
